# 0.1函数调用方式

函数在 JavaScript 里有 **4 种常见调用方式**，只有第一种是“被对象调用”：

表格复制

| 调用方式                     | this 指向            | 例子                              |
| :----------------------- | :----------------- | :------------------------------ |
| 1. 方法调用                  | 点前面的对象             | `obj.fn()` → `this === obj`     |
| 2. 普通函数调用                | 全局（严格模式 undefined） | `fn()` → `this === window`      |
| 3. 构造函数调用                | 新实例                | `new fn()` → `this === 实例`      |
| 4. 间接调用（call/apply/bind） | 手动指定               | `fn.call(obj)` → `this === obj` |

***

### 日常代码里后 3 种更常见

```javascript
setTimeout(callback, 100);   // 普通调用
Promise.then(res => {});     // 普通调用
addEventListener('click', handler); // 浏览器内部帮你 call
```

因此：**函数不一定非要被对象调用，大部分场景是“裸调”或“被框架/系统间接调用”。**\
只有当你需要**稳定访问某个对象的属性**时，才用 `bind` 把它“绑”到对象身上。

### 4.手写深拷贝

PS:浅拷贝也可以用一样的模板，当然深拷贝考得多

```javascript
/*
思路：
1.用hash = new WeakMap()，防止循环引用
2.基本类型或者函数，直接返回
3.新建一个对象空壳
4.hash.set(origin, obj);提前设置防止无限递归
6.只拷贝对象自身的属性
5.递归调用赋值
6.返回
*/

function exten(origin, deep = true, hash = new WeakMap()) {
    if (hash.has(origin))
        return hash.get(origin);
    if (origin === null || typeof origin !== 'object') return origin;
    if (origin instanceof Date) return new Date(origin);
    if (origin instanceof RegExp) return new RegExp(origin);
    if (typeof origin === 'function') return origin;

    let obj = origin instanceof Array ? [] : {};
    hash.set(origin, obj);
    for (let key in origin) {
        if (!origin.hasOwnProperty(key)) continue;
        let val = origin[key];
        obj[key] = deep && typeof val === 'object' && val !== null ? exten(val, deep, hash) : val;
    }
    return obj;
}

console.log(n.self === n); // true
```

### 5.手写防抖节流

```javascript
/*
    思路：不是 debounce 被重复执行，而是它一次性生出来的那个闭包函数被事件系统反复调用；闭包把 timer 变量“锁”在自己怀里，因此每次都能访问到上一次留下的定时器 ID。
    防抖：
    节流：
*/


//防抖---最后一次触发后等待delay毫秒之后执行
function debounce(func, delay) {
    let timer = null;
    return function (...args) {
        if (timer) clearTimeout(timer);
        timer = setTimeout(() => {
            func.call(this, ...args);
        }, delay)
    }
}

//节流--delay毫秒内最多只触发一次
function throttle	(func, delay) {
    let timer = null;
    return function (...args) {
        if (!timer) {
            timer = setTimeout(() => {
                func.call(this, ...args);
                timer = null;
            }, delay);
        }
    }
}
```

### 6.手写Ajax请求

```javascript
/*
梳理一下逻辑：
1.首先返回一个promise对象目的是为了进行异步操作后，拿到返回的结果，.then(data=>{})中的就是返回的结果，.catch中的error就是错误的结果，
2.然后我创建一个xhr对象，用来进行发送请求的操作，
3.创建好了首先设置发送请求的类型，
4.先监听状态变化,再发送请求
5.然后发送请求之后回来执行回调，===4继续执行，如果状态码200发送成功拿到结果，否则失败
*/

function ajax(url) {
    return new Promise((resolve, reject) => {
        const xhr = new XMLHttpRequest;
        //指定请求类型
        xhr.open('GET', url, true);
        xhr.onreadystatechange = function () {
            if (xhr.readyState === 4) {
                if (xhr.status === 200) {
                    //回调
                    resolve(JSON.parse(xhr.responseText));
                }
                else {
                    reject('error');
                }
            }
        }
        //发送请求
        xhr.send(null);
    })
}
```

```javascript
    let backarr = [];
    while (array.length) {
        const index = parseInt(Math.random() * array.length);
        backarr.push(array[index]);
        array.splice(index, 1);
    }
    return backarr;

    for (i = array.length - 1; i > 0; i--) {
        const j = Math.random() * (i + 1);
        [array[i], array[j]] = [array[j], array[i]];
```

```javascript
//Promise.all
function myAll(promises) {
    return new Promise((resolve, reject) => {
        const result = [];
        let counter = 0;
        for (let i = 0; i < promises.length; i++) {
            Promise.resolve(promises[i]).then(res => {
                result[i] = res;

                }

        }
    })

        })
    });
}
```

```javascript
//手撕快排
//思路：“双指针从两头向中间扫，右找小左找大，相遇就把基准落位，再对左右子区间递归。”
const _quickSort = (array) => {
    quickSort(array, 0, array.length - 1);
    return array;
}

const quickSort = (array, start, end) => {
    if (end - start < 1)
        return;
    let left = start;
    let right = end;

    const base = array[left];

    while (left < right) {
        //从右向左找到小于基准值的值，放到左边
        while (left < right && array[right] >= base) right--;
        array[left] = array[right];

        //从左向右找到大于基准值的值，放到右边
        while (left < right && array[left] <= base) left++;
        array[right] = array[left];
    }
    //相遇落地基准值--这个时候你就会发现基准值左边的都小于base，右边都大于base
    array[left] = base;
    //将左右两边继续递归--末尾和开头就从基准值往左右移动一位
    quickSort(array, start, left - 1)//递归左边
    quickSort(array, left + 1, end)//递归右边

    return array
}
```

```javascript
function search(nums,target){
	let left = 0;
	let right = nums.length - 1;
	while(left <= right){
	let mid = Math.floor((left + right)/2);

	return -1;//没找到
```

```javascript
柯里化 = 把“一次传完所有参数”的函数，
变成“可以分多次传参”的函数，只有累计参数 ≥ 原函数所需个数时，才真正执行。
思路：
新函数被调用 → 重新进入 curried → 再判够不够 → 不够再返回新函数，形成递归攒参链
*/

function sum(x, y, z) {
    return x + y + z;
}

function hyCurrying(fn) {
    // 判断当前已经接收的参数的个数，和函数本身需要接收的参数是否一致
    function curried(...args) {
        // 1.当已经传入的参数 大于等于 需要的参数时，就执行函数
        if (args.length >= fn.length) {
            // 如果调用函数时指定了this，要将其绑定上去
            return fn.apply(this, args);
        }
        else {
            // 没有达到个数时，需要返回一个新的函数，继续来接收参数
            return function (...args2) {
                //return curried.apply(this, [...args, ...args2])
                // 接收到参数后，需要递归调用 curried 来检查函数的个数是否达到
                return curried.apply(this, args.concat(args2));
            }
        }
    }
    return curried;
}

var curryAdd = hyCurry(sum);

curryAdd(10, 20, 30);
curryAdd(10, 20)(30);
curryAdd(10)(20)(30);

```

记忆：**“grid 两列，第二列 1fr 自适应。”**

### 二、三栏布局（左右固定 200px，中间自适应）

#### 1. flex（现代首选）

```css
.container { display: flex; }
.left, .right { width: 200px; flex-shrink: 0; }
.center { flex: 1; }
```

记忆：**“父 flex，左右定宽，中 flex:1 吃剩余。”**

#### 2. grid（最简短）

```css
.container { display: grid; grid-template-columns: 200px 1fr 200px; }
```

记忆：**“grid 三列，中间 1fr 自适应。”**

#### 3. 圣杯布局（float 经典，面试爱考）

```html
<div class="container">
  <div class="center">中间</div>
  <div class="left">左</div>
  <div class="right">右</div>
</div>
```

```css
.container { padding: 0 200px; } /* 左右预留 */
.left, .right, .center { float: left; height: 100px; }
.center { width: 100%; }
.left { margin-left: -100%; position: relative; right: 200px; }
.right { margin-left: -200px; }
```

记忆：**“中 100%，左右用负 margin 拉回，父 padding 留位置。”**

***

### 三、一句话选型

*   **现代项目** → **flex 或 grid**（简洁、响应式、无浮动副作用）。
*   **兼容老 IE** → **float + margin / 圣杯**。

背口诀：

> **“现代 flex/grid，老项目 float，负 margin 圣杯面试常考。”**

```javascript
function myInterval(fn, time) {
    let timer = null;

    const interval = () => {
        timer = setTimeout(interval, time);
        fn();
    }

    timer = setTimeout(interval, time);

    return () => {
        clearTimeout(timer);

    }
}
```

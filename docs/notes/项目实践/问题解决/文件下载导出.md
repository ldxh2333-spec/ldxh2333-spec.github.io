// 下载文件
export function downloadFileByPath(path, name, suffix) {
  // 1. 创建一个<a>元素，用于触发浏览器原生下载
  const link = document.createElement('a')

  // 2. 隐藏元素，避免在页面上闪现
  link.style.display = 'none'

  // 3. 把文件地址赋给href（支持http/https/相对路径）
  link.href = path

  // 4. 拼出带时间戳的文件名：2025-12-30-文件名.后缀
  const fileName = parseTime(new Date()) + '-' + name + '.' + suffix

  // 5. 设置download属性，告诉浏览器“直接下载”并指定默认保存名
  link.setAttribute('download', fileName)

  // 6. 把<a>临时放进DOM，否则某些浏览器不响应click
  document.body.appendChild(link)

  // 7. 模拟用户点击，立即弹出下载框
  link.click()

  // 8. 下载触发后立刻移除元素，干净无残留
  document.body.removeChild(link)
}
